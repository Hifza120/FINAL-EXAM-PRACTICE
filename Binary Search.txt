#include<iostream>
using namespace std;

int main(){
    
    int arr[9]={1,2,3,4,5,6,7,8,9};
    int start=0;
    int end=9;
    int key=2;
    while(start<=end){
        int mid=(start+end)/2;
        if(arr[mid]==key){
            cout<<"Found at index:"<<mid<<endl;
            break;
        }
        else if(arr[mid]>key){
            end=mid-1;
        }
        else{
            start=mid+1;
        }
    }
    return 0;
}

#include<iostream>
using namespace std;
class Queue{
 protected:
 int *arr;
 int size;
 int capacity;
 public:
 Queue(int s){
     size=s;
     arr=new int[size];
     capacity=0;
     
 }
 void enqueue(int a){
     if(capacity==size){
         cout<<" Queue is Full "<<endl;
     }
     else{
         arr[capacity++]=a;
     }
 }
    
int dequeue(){
         if(capacity==0){
         cout<<" Queue is Empty"<<endl;
         return -1;
     }
     else{
         int data=arr[0];
         for(int i=0;i<capacity-1;i++){
             arr[i]=arr[i+1];
         }
         capacity--;
         return data;
     }

}
int peek(){
         if(capacity==0){
         cout<<" Queue is Empty"<<endl;
         return -1;
     }
     else{
         int data=arr[0];
         return data;
     }

}

bool isEmpty(){
    return capacity==0;
}

void display(){
    for(int i=0;i<capacity;i++){
        cout<<arr[i]<<endl;
    }
}
};

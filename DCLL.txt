#include<iostream>
using namespace std;
struct Node {
    int data;
    Node* next;
    Node* prev;

};
class DoublyCircularLinkedListADT {
protected:
    Node* tail;
public:
    DoublyCircularLinkedListADT() {
        tail = nullptr;
    }
    virtual bool isEmpty() = 0;
    virtual void insertathead(int value) = 0;
    virtual void insertattail(int value) = 0;
    virtual int deleteFromtail() = 0;
    virtual int deleteFromhead() = 0;
    virtual void displayFromhead() = 0;
    virtual void displayFromtail() = 0;
    
};
class DoublyLinkedList :public DoublyCircularLinkedListADT {
    Node* KTHLASTELEMENT(Node* t, int k) {
        static int count = 0;
        if (k == count) {
            return t;
        }
        count++;

        Node*res= KTHLASTELEMENT(t->prev, k);
        if (res != nullptr) {
            return res;
        }
        if (t->prev==tail) {
            return nullptr; //BASE CASE WE DID NOT FOUND
        }
        return nullptr;

    }
public:

    void insertatNode(int position,int data) {
        Node* nn = new Node;
        nn->data = data;
        nn->next = nullptr;
        nn->prev = nullptr;
        if (isEmpty()) {
            tail = nn;
            nn->next = tail;
            nn->prev = tail;
        }
        else if (tail==tail->next) {
            if (position == 1) {
                tail->next = nn;
                tail->prev = nn;
                nn->next = tail;
                nn->prev = tail;
            }
            if (position == 2) {
                nn->next = tail->next;
                nn->prev = tail->prev;
                tail = nn;
            }

        }
        else {
            Node* t=tail->next;
            int p = 1;
            bool flag = false;
            while (1) {
                if (p == position) {
                    flag = true;
                    break;
                }
                p++;
                t = t->next;
                if (t== tail->next) {

                    break;
                }
            }
            if (flag) {

                nn->next = t;
                nn->prev = t->prev;
                t->prev->next = nn;
                t->prev = nn;
            }
            else {
                nn->next = t;
                nn->prev = t->prev;
                t->prev->next = nn;
                t->prev = nn;
                tail = nn;

            }

        }
    }
    bool isEmpty() {
        return tail == nullptr;
    }
    void insertathead(int value) {
        Node* nn = new Node;
        nn->data = value;
        nn->next = nullptr;
        nn->prev = nullptr;
        if (isEmpty()) {
            tail = nn;
            tail->next = nn;
            tail->prev = nn;
        }
        else {
            nn->next = tail->next;
            nn->prev = tail;
            tail->next->prev = nn;
            tail->next = nn;
        }
    }
    void insertattail(int value) {
        Node* nn = new Node;
        nn->data = value;
        nn->next = nullptr;
        nn->prev = nullptr;
        if (isEmpty()) {
            tail = nn;
            tail->next = nn;
            tail->prev = nn;
        }
        else {
            nn->next = tail->next;
            nn->prev = tail;
            tail->next->prev = nn;
            tail->next = nn;
            tail = nn;
        }

    }
    int deleteFromtail() {
        if (isEmpty()) {
            return -1;
        }
        else if (tail == tail->next) {
            int data = tail->data;
            delete tail;
            tail = nullptr;
            return data;
        }
        else {
            Node* tprev = tail->prev;
            int data = tail->data;
            tprev->next = tail->next;
            tail->next->prev = tprev;
            delete tail;
            tail = tprev;
            return data;

        }
    }
    int deleteFromhead() {
        if (isEmpty()) {
            return -1;
        }
        else if (tail == tail->next) {
            int data = tail->data;
            delete tail;
            tail = nullptr;
            return data;
        }
        else {
            Node* temp = tail->next;
            int data = temp->data;
            tail->next = temp->next;
            delete temp;
            temp = nullptr;
            tail->next->prev = tail;
            return data;

        }

    }
    void displayFromhead() {
        if (isEmpty()) {
            cout << "Is Empty" << endl;
        }
        else {
            Node* t = tail->next;
            while (1) {
                cout << t->data << endl;
                t = t->next;
                if (t == tail->next) {
                    break;
                }
            }
        }
    }
    void displayFromtail() {
        if (isEmpty()) {
            cout << "Is Empty" << endl;
        }
        else {
            Node* t = tail;
            while (1) {
                cout << t->data << endl;
                t = t->prev;
                if (t == tail) {
                    break;
                }
            }
        }

    }
    bool reverseKGroups(int k) {
        if (isEmpty()) {
            return false;
        }
        if (tail == tail->next) {
            return false;
        }
        int size = 0;
        Node* t = tail->next;
        while (1) {
            size++;
            t = t->next;
            if (t == tail->next) {
                break;
            }
        }
        if (k >= size) {
            return false;
        }
        DoublyLinkedList D;

        while (!isEmpty()) {
            int count =1;
            DoublyLinkedList temp;
            while (count <= k && !isEmpty()) {
                temp.insertathead(deleteFromhead());
                count++;
            }
            while (!temp.isEmpty()) {
                D.insertattail(temp.deleteFromhead());
            }
        }
        while (!D.isEmpty()) {
            insertattail(D.deleteFromhead());
        }
        return true;
    }
    bool reverseALTERNATEKGroups(int k) {
        if (isEmpty()) {
            return false;
        }
        if (tail == tail->next) {
            return false;
        }
        int size = 0;
        Node* t = tail->next;
        while (1) {
            size++;
            t = t->next;
            if (t == tail->next) {
                break;
            }
        }
        if (k >= size) {
            return false;
        }
        DoublyLinkedList D;
        while (!isEmpty()) {
            int count = 1;
            DoublyLinkedList temp;
            while (count <= k && !isEmpty()) {
                temp.insertathead(deleteFromhead());
                count++;
            }
            while (!temp.isEmpty()) {
                D.insertattail(temp.deleteFromhead());
            }
            count = 1;
            while (count<=k&&!isEmpty()) {
                D.insertattail(deleteFromhead());
                count++;
            }
        }
        while (!D.isEmpty()) {
            insertattail(D.deleteFromhead());
        }
        return true;
    }
    bool reverse() {
        if (isEmpty()) {
            return false;
        }
        else  if (tail == tail->next) {
            return true;
        }
        else {
            Node* t = tail->next;
            Node* tailsave = tail->next;
            while (true) {
                Node* temp = t->prev;
                t->prev = t->next;
                t->next = temp;
                t = t->prev;
                if (t==tailsave) {
                    break;
                }
            }
            tail = tailsave;
            return true;
        }
    }

    bool sort() {
        if (isEmpty()) {
            return true;
        }
        if (tail == tail->next) {
            return true;
        }
        else {
            Node* t = tail->next;
            while (true) {
                Node* NEXT = t->next;
                Node* tnext = t->next;
                Node* temp =t;
                while (tnext != tail->next) {
                    if (temp->data > tnext->data) {
                        temp = tnext;
                    }
                    tnext = tnext->next;
                }
                if (temp != nullptr) {
                    if (t == tail) {
                        tail = temp;
                    }
                    if (temp == tail) {
                        tail = t;
                    }
                    Node* tempN = temp->next;
                    Node* tempP = temp->prev;
                    Node* tN = t->next;
                    Node* tP = t->prev;
                    tempN->prev = t;
                    tempP->next = t;
                    tN->prev = temp;
                    tP->next = temp;
                    t->next = tempN;
                    t->prev = tempP;
                    temp->next = tN;
                    temp->prev = tP;

                }
                t = NEXT;
                if (t == tail->next) {
                    break;
                }
            }
            return true;

        }
    }
    int triplets(int x) {
        if (isEmpty()) {
            return 0;
        }
        if (tail == tail->next) {
            if (x == tail->data) {
                return 1;
            }
            return 0;
        }
        else {
            int num = 0;
            Node* t = tail->next;

                Node* t2 = t->next;

                        t3 = t3->next;
                    }
                    t2 = t2->next;
                }
                t = t->next;
                if (t == tail->next) {
                    return num;
                    break;
                }
            }
            return num;
        }
    }

};

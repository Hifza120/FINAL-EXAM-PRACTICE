
#include<iostream>
using namespace std;
class stack{
  protected:
  int * arr;
  int capacity;
  int size;
  public:  
  stack(int s):size(s){
      arr=new int[size];
      capacity=0;
  }
  int top(){
      if(capacity==0){
          return -1;
      }
      else{
          return  arr[capacity-1];
      }
  }
  int  pop(){
            if(capacity==0){
          return -1;
      }
      else{
          int data=arr[capacity-1];
          capacity--;
          return  data;
      }

  }
  
  bool isEmpty(){
      return capacity==0;
  }
  void display(){
      for(int i=0;i<capacity;i++){
          cout<<arr[i]<<endl;
      }
  }
 void push(int value)
{
    if(size==capacity){
        cout<<"Size is Full"<<endl;
    }
    else{
        arr[capacity++]=value;
        
    }
} 
};

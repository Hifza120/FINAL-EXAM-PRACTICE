#include<iostream>
using namespace std;
struct Node {
    int data;
    Node* next;
    Node* prev;

};
class DoublyLinkedListADT {
protected:
    Node* head;
    Node* tail;
public:
    DoublyLinkedListADT() {
        head = nullptr;
        tail = nullptr;
    }
    virtual bool isEmpty() = 0;
    virtual void insertathead(int value) = 0;
    virtual void insertattail(int value) = 0;
    virtual int deleteFromtail() = 0;
    virtual int deleteFromhead() = 0;
    virtual void displayFromhead() = 0;
    virtual void displayFromtail() = 0;

};
class DoublyLinkedList :public DoublyLinkedListADT {
public:
    bool isEmpty() {
        return head == nullptr && tail == nullptr;
    }
    void insertathead(int value) {
        Node* nn = new Node;
        nn->data = value;
        nn->next = nullptr;
        nn->prev = nullptr;
        if (isEmpty()) {
            head = nn;
            tail = nn;
        }
        else {
            head->prev = nn;
            nn->next = head;
            head = nn;
        }
    }
    void insertattail(int value) {
        Node* nn = new Node;
        nn->data = value;
        nn->next = nullptr;
        nn->prev = nullptr;
        if (isEmpty()) {
            head = nn;
            tail = nn;
        }
        else {
            tail->next = nn;
            nn->prev = tail;
            tail = nn;
        }

    }
    int deleteFromtail() {
        if (isEmpty()) {
            return -1;
        }
        else if (head == tail) {
            int data = head->data;
            delete head;
            head = nullptr;
            tail = nullptr;
            return data;
        }
        else {
            int data = tail->data;
            tail = tail->prev;
            delete tail->next;
            tail->next = nullptr;
            return data;

        }
    }
    int deleteFromhead() {
        if (isEmpty()) {
            return -1;
        }
        else if (head == tail) {
            int data = head->data;
            delete head;
            head = nullptr;
            tail = nullptr;
            return data;
        }
        else {
            int data = head->data;
            head = head->next;
            delete head->prev;
            head->prev = nullptr;
            return data;

        }

    }
    void displayFromhead() {
        if (isEmpty()) {
            cout << "Is Empty" << endl;
        }
        else {
            Node* t = head;
            while (1) {
                cout << t->data << endl;
                t = t->next;
                if (t == nullptr) {
                    break;
                }
            }
        }
    }
    void displayFromtail() {
        if (isEmpty()) {
            cout << "Is Empty" << endl;
        }
        else {
            Node* t = tail;
            while (1) {
                cout << t->data << endl;
                t = t->prev;
                if (t == nullptr) {
                    break;
                }
            }
        }

    }
    void insertSorted(int data) {
        if (isEmpty()) {
            Node* nn = new Node;
            nn->data = data;
            nn->next = nullptr;
            nn->prev = nullptr;
            head = nn;
            tail = nn;
        }
        else if (data < head->data) {
            insertathead(data);
        }
        else if (data > tail->data) {
            insertattail(data);
        }
        else {
            Node* nn = new Node;
            nn->data = data;
            nn->next = nullptr;
            nn->prev = nullptr;

            Node* t = head;
            while (true) {
                if (t->next != nullptr) {
                    if (t->data < data && t->next->data >= data) {
                        nn->prev = t;
                        nn->next = t->next;
                        t->next = nn;
                        nn->next->prev = nn;

                    }
                }
                t = t->next;
                if (t == nullptr) {
                    break;
                }
            }
        }

    }
    void insertATposition(int position, int data) {
        if (isEmpty()) {
            Node* nn = new Node;
            nn->data = data;
            nn->next = nullptr;
            nn->prev = nullptr;
            head = nn;
            tail = nn;
        }
        else if (tail == head) {
            if (position == 1) {
                insertathead(data);
            }
            else {
                insertattail(data);
            }
        }
        else {
            int p = 1;
            Node* t = head;
            while (t != nullptr) {
                if (p == position) {
                    break;
                }
                p++;
                t = t->next;
            }
            if (t == head) {
                insertathead(data);
            }
            if (t == nullptr) {
                insertattail(data);
            }
            else {
                Node* nn = new Node;
                nn->data = data;
                nn->next = t;
                nn->prev = t->prev;
                nn->prev->next = nn;
                t->prev = nn;

            }
        }
    }
    bool DeleteATVALUE(int value) {
        if (isEmpty()) {
            return true;
        }
        if (tail == head) {
            if (tail->data == value) {
                deleteFromtail();
                return true;
            }
            return false;
        }
        if (tail->data == value) {
                deleteFromtail();
                return true;
        }
        if (head->data == value) {
            deleteFromhead();
            return true;
        }
        else {
            Node* t = head;
            while (t != nullptr) {
                if (t->data == value) {
                    t->prev->next = t->next;
                    t->next->prev = t->prev;
                    delete t;
                    t = nullptr;
                    return true;
                }
                t = t->next;
            }
            return false;
        }
      
    }
    void merge2list( DoublyLinkedList& L1) {
        while (!L1.isEmpty()) {
            insertattail(L1.deleteFromhead());
        }
    }
    void mergeKlist(DoublyLinkedList D[], int k) {
        for (int i = 0; i < k; i++) {
            merge2list(D[i]);
        }
    }
};

int main() {
    DoublyLinkedList dll;

    cout << "Testing insertathead and insertattail:" << endl;
    dll.insertathead(10);
    dll.insertathead(5);
    dll.insertattail(20);
    dll.insertattail(25);
    cout << "List from head:" << endl;
    dll.displayFromhead();
    cout << "List from tail:" << endl;
    dll.displayFromtail();

    cout << "\nTesting deletion from head and tail:" << endl;
    int headVal = dll.deleteFromhead();
    cout << "Deleted from head: " << headVal << endl;
    int tailVal = dll.deleteFromtail();
    cout << "Deleted from tail: " << tailVal << endl;
    cout << "List after deletions:" << endl;
    dll.displayFromhead();

    cout << "\nTesting insertSorted:" << endl;
    dll.insertSorted(15); // Should insert between 10 and 20
    dll.insertSorted(2);  // Should become new head
    dll.insertSorted(30); // Should become new tail
    cout << "List after sorted insertions:" << endl;
    dll.displayFromhead();

    cout << "\nTesting insertATposition (inserting 12 at position 3):" << endl;
    dll.insertATposition(3, 12);
    dll.displayFromhead();

    cout << "\nTesting DeleteATVALUE (deleting value 15):" << endl;
    if (dll.DeleteATVALUE(15))
        cout << "Value 15 deleted successfully." << endl;
    else
        cout << "Value 15 not found." << endl;
    dll.displayFromhead();

    cout << "\nTesting merge2list:" << endl;
    DoublyLinkedList dll2;
    dll2.insertattail(40);
    dll2.insertattail(50);
    dll2.insertattail(60);
    dll.merge2list(dll2);
    cout << "List after merging second list:" << endl;
    dll.displayFromhead();

    return 0;
}

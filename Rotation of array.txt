#include<iostream>
using namespace std;

int main(){
    
    int arr[9]={1,2,3,4,5,6,7,8,9};
    int k=2;
    int size=9;
    for(int i=0;i<k;i++){
    
        int temp=arr[size-1];
        for(int j=size-1;j>0;j--){
          arr[j]=arr[j-1];
            
        }
        arr[i]=temp;
    }
    for(int i=0;i<size;i++){
        cout<<arr[i]<<"\t";
    }
    return 0;
}

#include<iostream>
using namespace std;
struct Node{
    int  data;
    Node*next;
};
class LinkedList{
  protected:
  Node*head;
  Node*tail;
  public:
  LinkedList(){
      head=nullptr;
      tail=nullptr;
  }
  bool isEmpty(){
      return head==nullptr&&tail==nullptr;
  }
  void insertattail(int value){
      Node*nn=new Node;
      nn->data=value;
      nn->next=nullptr;
      if(isEmpty()){
         head=nn;
         tail=nn;
      }
      else if(head==tail){
        tail->next=nn;
        tail=nn;
      }
      else{
      tail->next=nn;
        tail=nn;

      }
  }
    void insertathead(int value){
      Node*nn=new Node;
      nn->data=value;
      nn->next=nullptr;
      if(isEmpty()){
         head=nn;
         tail=nn;
      }
      else{
           nn->next=head;
           head=nn;

      }
  }
  int deleteFromtail(){
      if(isEmpty()){
          return NULL;
      }
      else if(head==tail){
          int data=head->data;
          delete head;
          head=nullptr;
          tail=nullptr;
          return data;
      }
      else{
          Node*t=head;
          while(1){
              t=t->next;
              if(t->next==tail){
                  break;
              }
          }
          t->next=nullptr;
          int data=tail->data;
          delete tail;
          tail=t;
          return data;
      }
  }
    int deleteFromhead(){
      if(isEmpty()){
          return NULL;
      }
      else if(head==tail){
          int data=head->data;
          delete head;
          head=nullptr;
          tail=nullptr;
          return data;
      }
      else{
          int data=head->data;
          Node*t=head->next;
          delete head;
          head=t;
          return data;
      }
  }
void display(){
    if(isEmpty()){
        cout<<" is Empty"<<endl;
    }
    else{
        Node*t=head;
        while(1){
            cout<<t->data<<endl;
            t=t->next;
            if(t==nullptr){
                break;
            }
        }
    }
}

};
